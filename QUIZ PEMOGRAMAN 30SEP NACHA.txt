JAWABAN SOAL NO.5 : 
# Daftar produk (1-5)
produk = {
    1: ("Air Mineral", 15000),
    2: ("Teh Botol", 18000),
    3: ("Kopi Susu", 20000),
    4: ("Cokelat", 22000),
    5: ("Roti Sobek", 25000)
}


keranjang = []
total_tagihan = 0


print("=== VENDING MACHINE ===")
while True:
    print("\nDaftar Produk:")
    for id_p, (nama, harga) in produk.items():
        print(f"{id_p}. {nama} - Rp{harga:,}")
    print("6. Selesai Memilih")
    
    pilihan = int(input("\nPilih menu (1-6): "))
    
    if pilihan == 6:
        print("\nAnda memilih untuk menyudahi transaksi.")
        break
    elif pilihan in produk:
        nama, harga = produk[pilihan]
        
        # Cek syarat: maksimal 3 produk ATAU total tagihan mencapai/melebihi Rp50.000
        keranjang.append((nama, harga))
        total_tagihan += harga
        
        print(f"-> {nama} berhasil ditambahkan.")
        
        if len(keranjang) >= 3 or total_tagihan >= 50000:
            print("\nSyarat pembelian terpenuhi (Maksimal 3 produk / Total >= Rp50.000)!")
            break
    else:
        print("Pilihan tidak valid, silakan coba lagi.")


# Rincian Transaksi
print("\n=== RINGKASAN PEMBELIAN ===")
if keranjang:
    for item, harga in keranjang:
        print(f"- {item}: Rp{harga:,}")
    print(f"Total Tagihan: Rp{total_tagihan:,}")
else:
    print("Tidak ada produk yang dibeli.")
